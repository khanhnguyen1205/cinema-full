- [ ] **Step 3: Viết init** — `src/i18n/index.ts`

```ts
import i18n from "i18next";
import { initReactI18next } from "react-i18next";
import LanguageDetector from "i18next-browser-languagedetector";
import vi from "./locales/vi.json";
import en from "./locales/en.json";

i18n
  .use(LanguageDetector)
  .use(initReactI18next)
  .init({
    resources: {
      vi: { translation: vi },
      en: { translation: en },
    },
    fallbackLng: "vi",
    supportedLngs: ["vi", "en"],
    // CHỈ localStorage + htmlTag, KHÔNG navigator -> mặc định vi, e2e xanh
    detection: {
      order: ["localStorage", "htmlTag"],
      caches: ["localStorage"],
      lookupLocalStorage: "lang",
    },
    interpolation: { escapeValue: false }, // React đã escape
  });

// Đồng bộ <html lang> theo ngôn ngữ hiện tại
document.documentElement.lang = i18n.language || "vi";
i18n.on("languageChanged", (lng) => {
  document.documentElement.lang = lng;
});

export default i18n;
```

- [ ] **Step 4: Viết format helper** — `src/i18n/format.ts`

```ts
import i18n from "i18n";

const localeOf = (): string =>
  (i18n.language || "vi").startsWith("en") ? "en-US" : "vi-VN";

// Giá LUÔN VND, chỉ nhóm số theo locale
export const formatPrice = (n: number): string =>
  `${n.toLocaleString(localeOf())} ₫`;

export const formatDateTime = (
  iso: string,
  opts?: Intl.DateTimeFormatOptions,
): string => new Date(iso).toLocaleString(localeOf(), opts);

export const formatDate = (
  iso: string,
  opts?: Intl.DateTimeFormatOptions,
): string => new Date(iso).toLocaleDateString(localeOf(), opts);
```

          type="button"

          aria-pressed={current === l.code}
          onClick={() => i18n.changeLanguage(l.code)}
        >
          {l.label}
        </button>
      ))}
    </div>
  );
}

- [ ] **Step 1: Bảng thể loại** — `src/i18n/genres.ts`

```ts
import i18n from "i18n";

// Nhãn hiển thị thể loại theo ngôn ngữ; fallback trả về mã gốc nếu thiếu khoá.
export function genreLabel(code: string): string {
  const key = `genres.${code}`;
  const label = i18n.t(key);
  return label === key ? code : label;
}
```

Thêm vào `vi.json`/`en.json` khoá `genres` cho MỌI mã có trong seed. Lấy danh sách mã:

```bash
grep -oE '"genre": *"[^"]+"' db.json | sort -u
```

Ví dụ (điền đủ theo kết quả grep): `vi.json` → `"genres": { "Hành động": "Hành động", ... }` — **lưu ý:** genre trong db.json là chuỗi như `Action`/`Sci-Fi` hay đã tiếng Việt?
